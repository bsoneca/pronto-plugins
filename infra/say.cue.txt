// The infra verbs (README.md): the state in CUE, applied by OpenTofu.
//   ./saytw setup          the tools, the bucket OpenTofu keeps its state in, and tofu init
//   ./saytw generate       tofu/main.tf.json from the state (cue export -e plan.tofu)
//   ./saytw lint           cue vet, cue fmt, and tofu validate
//   ./saytw verify@infra   the plan: what applying would change, kept for release; fails when
//                          the live projects differ from the state, and refuses a plan that deletes
//   ./saytw release@infra  applies the plan verify@infra kept, read again first, and nothing else
// Each runs as the operator's own gcloud login, the gcloud on their PATH.
package say

_tofu: "use tools.nu [run-mise]; with-env {GOOGLE_OAUTH_ACCESS_TOKEN: (^gcloud auth print-access-token | str trim)} { run-mise exec -- tofu -chdir=tofu"
_backend: "(run-mise exec -- cue export . -e plan.backend --out json | from json)"

// Reads a kept plan and refuses one that cannot be read or that deletes anything. Dropping a
// resource from the state is no reason to delete it from the project: INFRA_ALLOW_DESTROY=1, for
// that run alone, says it is.
_check: "def check-plan [f: string] { let j = (run-mise exec -- tofu -chdir=tofu show -json $f | from json); if ($j | get -o format_version) == null { error make {msg: 'the plan could not be read'} }; let rc = ($j | get -o resource_changes | default []); if ($rc | describe) !~ '^(list|table)' { error make {msg: 'the plan lists its changes as no list'} }; let gone = ($rc | where {|c| 'delete' in $c.change.actions} | get address); if ($gone | is-not-empty) and ($env.INFRA_ALLOW_DESTROY? != '1') { error make {msg: $\"the plan deletes ($gone | str join ', '): set INFRA_ALLOW_DESTROY=1 to allow it\"} } }"

say: {
	setup: rulemap: tofu: {
		// Before sayt's own setup, which ends the verb once it has installed the tools.
		priority: -1
		stop:     true
		cmds: [
			{do: "use tools.nu [run-mise]; run-mise install"},
			{do: "use tools.nu [run-mise]; let s = \(_backend); let b = $\"gs://($s.bucket)\"; if (^gcloud storage buckets describe $b | complete).exit_code != 0 { ^gcloud storage buckets create $b --project $s.project --location $s.location --uniform-bucket-level-access --public-access-prevention; ^gcloud storage buckets update $b --versioning }"},
			{do: "use tools.nu [run-mise]; mkdir tofu; run-mise exec -- cue export . -e plan.tofu --out json | save -f tofu/main.tf.json"},
			{do: "\(_tofu) init -input=false }"},
		]
	}
	// sayt's own CUE generation would write over the tofu folder: this one exports plan.tofu.
	generate: rulemap: "auto-cue": null
	generate: rulemap: tofu: cmds: [{do: "use tools.nu [run-mise]; mkdir tofu; run-mise exec -- cue export . -e plan.tofu --out json | save -f tofu/main.tf.json"}]
	// sayt's own CUE lint would take the state for a schema of the tofu folder: this one vets the package.
	lint: rulemap: "auto-cue": null
	lint: rulemap: tofu: cmds: [
		{do: "use tools.nu [run-mise]; run-mise exec -- cue vet -c ."},
		{do: "use tools.nu [run-mise]; run-mise exec -- cue fmt --check ."},
		{do: "use tools.nu [run-mise]; run-mise exec -- tofu -chdir=tofu init -backend=false -input=false | ignore; run-mise exec -- tofu -chdir=tofu validate"},
	]
	// The plan is always of the state as it is now. Whatever plan was kept is removed first; the
	// whole state is vetted before it is exported (an export alone skips the schema's reasons);
	// the new plan is made apart and kept as infra.tfplan only once it has been read and passed.
	// It changes nothing in the projects: -detailed-exitcode says 2 when applying would.
	verify: rulemap: infra: {
		platform: "infra"
		stop:     true
		cmds: [
			{do: "rm -f tofu/infra.tfplan tofu/infra.tfplan.tmp"},
			{do: "use tools.nu [run-mise]; run-mise exec -- cue vet -c ."},
			{do: "use tools.nu [run-mise]; mkdir tofu; run-mise exec -- cue export . -e plan.tofu --out json | save -f tofu/main.tf.json"},
			{do: "use tools.nu [run-mise]; \(_check); with-env {GOOGLE_OAUTH_ACCESS_TOKEN: (^gcloud auth print-access-token | str trim)} { let r = (run-mise exec -- tofu -chdir=tofu plan -input=false -detailed-exitcode -out=infra.tfplan.tmp | complete); print $r.stdout; if $r.exit_code == 1 { rm -f tofu/infra.tfplan.tmp; error make {msg: $r.stderr} }; try { check-plan infra.tfplan.tmp } catch {|e| rm -f tofu/infra.tfplan.tmp; error make {msg: $e.msg} }; mv tofu/infra.tfplan.tmp tofu/infra.tfplan; if $r.exit_code == 2 { error make {msg: 'the live projects differ from the state: read the plan, then release@infra'} } }"},
		]
	}
	release: rulemap: infra: {
		platform: "infra"
		stop:     true
		cmds: [{do: "use tools.nu [run-mise]; \(_check); if not (\"tofu/infra.tfplan\" | path exists) { error make {msg: \"no plan: run ./saytw verify@infra first, and read it\"} }; with-env {GOOGLE_OAUTH_ACCESS_TOKEN: (^gcloud auth print-access-token | str trim)} { check-plan infra.tfplan; run-mise exec -- tofu -chdir=tofu apply -input=false infra.tfplan }; rm tofu/infra.tfplan"}]
	}
}
