// The infra verbs (README.md): the state in CUE, applied by OpenTofu.
//   ./saytw setup          the tools, the bucket OpenTofu keeps its state in, and tofu init
//   ./saytw generate       tofu/main.tf.json from the state (cue export -e plan.tofu)
//   ./saytw lint           cue vet, cue fmt, and tofu validate
//   ./saytw verify@infra   the plan: what applying would change, kept as tofu/plan-<sha256>.tfplan
//                          (plan.nu's saved-plan make) and its digest printed; fails when the live
//                          projects differ from the state, and refuses a plan that deletes
//   INFRA_PLAN_SHA256=<digest> ./saytw release@infra
//                          applies the plan of that digest, read again first, and nothing else
// Each runs as the operator's own gcloud login, the gcloud on their PATH.
package say

_tofu: "use tools.nu [run-mise]; with-env {GOOGLE_OAUTH_ACCESS_TOKEN: (^gcloud auth print-access-token | str trim)} { run-mise exec -- tofu -chdir=tofu"
_backend: "(run-mise exec -- cue export . -e plan.backend --out json | from json)"

// What a plan may delete is plan.nu's rule: nothing, unless INFRA_ALLOW_DESTROY=1 (exactly) is set for that run. A
// file's secret version replaced is no deletion: its policy is ABANDON, so the old version stays in Secret Manager.
_spare: "{|c| $c.type == 'google_secret_manager_secret_version' and ($c.change.before?.deletion_policy? == 'ABANDON')}"
_plan:  "use tools.nu [run-mise]; use cue.mod/pkg/github.com/bsoneca/pronto-plugins/infra/plan.nu *"
_run:   "{|a| run-mise exec -- tofu -chdir=tofu ...$a | complete}"
_creds: "{|| {GOOGLE_OAUTH_ACCESS_TOKEN: (gcloud-token)}}"
// A folder's own refusal of a plan (PRONTO-696): when the folder holds an executable-by-sh `infra-check`, it runs, under the
// folder's tools, after each plan is read and before it is kept (verify@infra) and again before it is applied (release@infra); a
// non-zero exit refuses the plan and its output is the reason. A folder without one is as it was.
_check: "{|rc| if ('infra-check' | path exists) { let r = (run-mise exec -- sh infra-check | complete); if $r.exit_code != 0 { error make {msg: $'infra-check refused the plan: ($r.stdout) ($r.stderr)'} } } }"

say: {
	setup: rulemap: tofu: {
		// Before sayt's own setup, which ends the verb once it has installed the tools.
		priority: -1
		stop:     true
		cmds: [
			{do: "use tools.nu [run-mise]; run-mise install"},
			{do: "\(_plan); state-bucket \(_backend) --gcloud {|a| ^gcloud ...$a | complete}"},
			{do: "use tools.nu [run-mise]; mkdir tofu; run-mise exec -- cue export . -e plan.tofu --out json | save -f tofu/main.tf.json"},
			{do: "\(_tofu) init -input=false }"},
		]
	}
	// sayt's own CUE generation would write over the tofu folder: this one exports plan.tofu.
	generate: rulemap: "auto-cue": null
	generate: rulemap: tofu: cmds: [{do: "use tools.nu [run-mise]; mkdir tofu; run-mise exec -- cue export . -e plan.tofu --out json | save -f tofu/main.tf.json"}]
	// sayt's own CUE lint would take the state for a schema of the tofu folder: this one vets the package.
	lint: rulemap: "auto-cue": null
	lint: rulemap: tofu: cmds: [
		{do: "use tools.nu [run-mise]; run-mise exec -- cue vet -c ."},
		{do: "use tools.nu [run-mise]; run-mise exec -- cue fmt --check ."},
		{do: "use tools.nu [run-mise]; run-mise exec -- tofu -chdir=tofu init -backend=false -input=false | ignore; run-mise exec -- tofu -chdir=tofu validate"},
	]
	// The plan is always of the state as it is now. The whole state is vetted before it is exported (an export alone
	// skips the schema's reasons); saved-plan make plans apart, reads and checks the plan, and keeps it only as
	// tofu/plan-<sha256>.tfplan, never removing a plan another verify kept. It changes nothing in the projects:
	// -detailed-exitcode says 2 when applying would.
	verify: rulemap: infra: {
		platform: "infra"
		stop:     true
		cmds: [
			{do: "use tools.nu [run-mise]; run-mise exec -- cue vet -c ."},
			{do: "use tools.nu [run-mise]; mkdir tofu; run-mise exec -- cue export . -e plan.tofu --out json | save -f tofu/main.tf.json"},
			{do: "\(_plan); saved-plan make --tofu \(_run) --creds \(_creds) --check \(_check) --spare \(_spare) | ignore"},
		]
	}
	// Applies the kept plan whose digest verify@infra printed, INFRA_PLAN_SHA256, and nothing else.
	release: rulemap: infra: {
		platform: "infra"
		stop:     true
		cmds: [{do: "\(_plan); saved-plan apply ($env.INFRA_PLAN_SHA256? | default '') --tofu \(_run) --creds \(_creds) --check \(_check) --spare \(_spare)"}]
	}
}
